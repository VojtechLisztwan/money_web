🛡️Povolení kolekcí ve Firebase

Ve vaší Firebase Console přejděte do Firestore Database ➔ záložka Rules (Pravidla) a vložte následující pravidla: 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // 1) Seznam povolených e-mailů (whitelist)
    function isWhitelisted() {
      return request.auth != null && 
             request.auth.token.email in [
               'vojtalisztwan@gmail.com',
               'test@test.cz',
               'david.krop.x@gmail.com'
             ];
    }

    // 2) Uživatel smí pracovat POUZE se svými daty
    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }

    match /transactions/{docId} {
      allow read, delete: if isWhitelisted() && isOwner(resource.data.userId);
      allow create: if isWhitelisted() && isOwner(request.resource.data.userId);
      allow update: if isWhitelisted() && isOwner(resource.data.userId) && isOwner(request.resource.data.userId);
    }

    match /work_projects/{docId} {
      allow read, delete: if isWhitelisted() && isOwner(resource.data.userId);
      allow create: if isWhitelisted() && isOwner(request.resource.data.userId);
      allow update: if isWhitelisted() && isOwner(resource.data.userId) && isOwner(request.resource.data.userId);
    }

    match /work_logs/{docId} {
      allow read, delete: if isWhitelisted() && isOwner(resource.data.userId);
      allow create: if isWhitelisted() && isOwner(request.resource.data.userId);
      allow update: if isWhitelisted() && isOwner(resource.data.userId) && isOwner(request.resource.data.userId);
    }

    match /vehicles/{docId} {
      allow read, delete: if isWhitelisted() && isOwner(resource.data.userId);
      allow create: if isWhitelisted() && isOwner(request.resource.data.userId);
      allow update: if isWhitelisted() && isOwner(resource.data.userId) && isOwner(request.resource.data.userId);
    }

    match /vehicle_events/{docId} {
      allow read, delete: if isWhitelisted() && isOwner(resource.data.userId);
      allow create: if isWhitelisted() && isOwner(request.resource.data.userId);
      allow update: if isWhitelisted() && isOwner(resource.data.userId) && isOwner(request.resource.data.userId);
    }

    match /vehicle_files/{docId} {
      allow read, delete: if isWhitelisted() && isOwner(resource.data.userId);
      allow create: if isWhitelisted() && isOwner(request.resource.data.userId);
      allow update: if isWhitelisted() && isOwner(resource.data.userId) && isOwner(request.resource.data.userId);
    }
  }
}

Po uložení v Firebase Console (tlačítko Publish) stačí obnovit stránku v prohlížeči (F5). Data se budou ukládat do Firebase cloudu a synchronizovat napříč všemi vašimi zařízeními.
